import numpy as np
import pandas as pd

def evaluate_bollinger_rsi_strategy(df: pd.DataFrame, window: int = 20, num_std: float = 2.0) -> pd.DataFrame:
    """Vectorized backtesting pipeline for Bollinger Bands, Moving Averages & RSI signals."""
    df = df.copy()
    # 1. Rolling Moving Average & Volatility Bands
    df['sma'] = df['close'].rolling(window=window).mean()
    df['std'] = df['close'].rolling(window=window).std(ddof=0)
    df['upper_band'] = df['sma'] + (num_std * df['std'])
    df['lower_band'] = df['sma'] - (num_std * df['std'])

    # 2. Relative Strength Index (RSI-14)
    delta = df['close'].diff()
    gain = delta.clip(lower=0).rolling(window=14).mean()
    loss = (-delta.clip(upper=0)).rolling(window=14).mean()
    rs = gain / loss.replace(0, np.nan)
    df['rsi'] = 100.0 - (100.0 / (1.0 + rs))

    # 3. Signal Generation & Position Shifting (No Lookahead Bias)
    df['signal'] = np.where(df['close'] < df['lower_band'], 1.0,
                   np.where(df['close'] > df['upper_band'], 0.0, np.nan))
    df['position'] = df['signal'].ffill().fillna(0.0).shift(1).fillna(0.0)

    # 4. Strategy Returns, Equity Curve & Sharpe Evaluation
    df['asset_ret'] = df['close'].pct_change().fillna(0.0)
    df['strat_ret'] = df['position'] * df['asset_ret']
    df['equity_curve'] = 100.0 * (1.0 + df['strat_ret']).cumprod()
    return df
